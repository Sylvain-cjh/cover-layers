cmake_minimum_required(VERSION 3.14)
project(MD3CoverLayerBackend LANGUAGES CXX RC)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The cover backend requires 64-bit Windows.")
endif()
add_library(cover_layers_gpu SHARED cover_layers_gpu.cpp cover_gpu_resources.rc)
set_source_files_properties(cover_gpu_resources.rc PROPERTIES OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/cover_gpu_compute.hlsl")
target_compile_features(cover_layers_gpu PRIVATE cxx_std_17)
target_compile_definitions(cover_layers_gpu PRIVATE NOMINMAX UNICODE _UNICODE)
target_include_directories(cover_layers_gpu SYSTEM PRIVATE include/onnxruntime include/directml)
target_link_libraries(cover_layers_gpu PRIVATE d3d12 dxgi dxcore d3dcompiler)
